DHAKA(config-if)#ip address 10.0.0.1 255.255.255.252
DHAKA(config-if)#tunnel mode gre ip
DHAKA(config-if)#tunnel source gigabitEthernet 0/1
DHAKA(config-if)#tunnel destination 201.0.0.2
DHAKA(config-if)#exit
