============DHAKA============
DHAKA#conf ter
DHAKA(config)#interface gigabitEthernet  0/1
DHAKA(config-if)#no shutdown 
DHAKA(config-if)#ip address 200.0.0.1 255.255.255.252
DHAKA(config-if)#exit

#Default Route Towards ISP_A
DHAKA(config)#ip route 0.0.0.0 0.0.0.0 200.0.0.2

#Config NAT(PAT)
DHAKA(config)#access-list 10 permit 192.168.10.0 0.0.0.255 
DHAKA(config)#access-list 10 permit 192.168.20.0 0.0.0.255 
DHAKA(config)#access-list 10 permit 192.168.30.0 0.0.0.255 
DHAKA(config)#ip nat pool natp1 200.0.0.1 200.0.0.1 netmask 255.255.255.252
DHAKA(config)#ip nat  inside source list 10 pool natp1 overload 

DHAKA(config)#interface gigabitEthernet 0/0
DHAKA(config-if)#ip nat inside 
DHAKA(config-if)#exit

DHAKA(config)#interface gigabitEthernet 0/1
DHAKA(config-if)#ip nat outside 
DHAKA(config-if)#exit

CUMILLA(config)#ip nat inside source list 20 pool pool1 overload

CUMILLA(config-subif)#ip nat inside
CUMILLA(config-subif)#exit

CUMILLA(config)#interface gigabitEthernet 0/0.20
CUMILLA(config-subif)#ip nat inside
CUMILLA(config-subif)#exit
